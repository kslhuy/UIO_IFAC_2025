\begin{abstract}
This paper presents a solution to estimate state of k-nearest neighbor in platoon. A bunch of technic is used to secured estimation in the case of measurement fautlt in local vehicle and cyber-attack in communication channel .
\end{abstract}

\begin{keywords}
Distributed observers, secure estimation, lyapunov stability, cyber-attack.
\end{keywords}

\section{Introduction}
    

The contributions of this paper are as follows:
\begin{enumerate}
\item On each vehicle, a local UIO is used for secure estimation of the state in the case measurement has a fault. 
\item A Distributed observer collected the local state estimation of a k-neighbor vehicle to build a bigger system that can estimate all the states of the neighbor vehicles.
\item Using some techniques like a neuro-adaptive observer for estimating the unknow uncertainty in the local vehicle, and PIO for estimation the difference between the local vehicle and other neighbor vehicles. 
\item (Not decided yet) A bank of observers to detect faults in measurement, and use global fusion for have a secure estimation in local vehicle 
\item (Not decided yet) Using L1-norm estimation for estimated the cyber-attack in the communications channels.

\protect\href{https://github.com/kslhuy/CACC_UIO_simulink}{GitHub}.
\end{enumerate}



\subsection{CACC mathematical model}
The dynamics of $i^{th}$ vehicle are a Position-Velocity-Acceleration third-order linear model given by the following set of equations:
\begin{equation}\label{eq_1}
      \left\{ \begin{array}{l}
        \dot{p}_i(t) =v_i(t) \\
        \dot{v}_i(t)  =a_i(t) + f_{vi}(v,a) \\
        \dot{a}_i(t)  =\frac{1}{\tau_i} u_i(t)-\frac{1}{\tau_i} a_i(t) + f_{ai}(v,a)  \end{array}  \right.
\end{equation}  
where
\begin{itemize}
\item $p_i(t), v_i(t)$ and $a_i(t)$ denote the position, velocity and
acceleration of the $i^{th}$  vehicle, respectively;
\item $u_i(t)$ is the $i^{th}$ vehicle control input representing the desired
driving/braking force or acceleration;
 \item $\tau$ is the engine's constant time lag. 
 \item $f_i$ is the nonlinear uncertainty 
 % \item $\Delta_i$ perturbation that is injected in the system. 
 \end{itemize}

 All the equations can written :

\begin{equation}\label{CACC}
%\displaystyle
       \begin{array}{l}
        {x_i}(t+1)=A x_i + B  (u_i + f(x)_i + \Delta_i)  \\
        y_{i}(t) = Cx_{i}\end{array} .
\end{equation}  

where 
\begin{equation*}
        A  =\begin{bmatrix}
        0 & 1 & 0 \\
        0 & 0 & 1 \\
        0 & 0 & \frac{-1}{\tau}
          \end{bmatrix},
        B =\begin{bmatrix}
        0 \\
        0 \\
        \frac{1}{\tau}
          \end{bmatrix}, 
        C =\begin{bmatrix}
        1 & 0 & 0 \\
        0 & 1 & 0 \\
        0 & 0 & 1
          \end{bmatrix}, 
\end{equation*}

Each vehicle has a full measurement state obtained in real-time \textbf{position} and \textbf{velocity} from a GPS mounted on the vehicle. 

\section{Neuro-adative observer}
We use the RBF neural network to approximate the nonlinear uncertainty in the system

\begin{equation}
    f(x,t)_i = W^T\Theta({x}_i) + \epsilon_i
\end{equation}

Let us assume $f(x,t)_i)$ can be approximated by
\begin{equation}
    \hat{f}(x,t)_i = \hat{W}^T\Theta(\hat{x}_i) 
\end{equation}


The UIO observer is given by: 
\begin{equation}\label{UIO}
    \left\{ \begin{array}{l}
        z_{t+1} = {N}z_{t}+ {L}y + {M}u + G\hat{f}\\ 
         \hat{x}_{t} = z_{t} + Qy
     \end{array}\right.
\end{equation}
Adaptation law is: 
\begin{align}
    \hat{W}(t) = \alpha \hat{W}(t-\tau) + \beta( \hat{W}(t-\tau) + \hat{W}(t-\tau + 1)) \\+\gamma((y - \hat{y})^T C {A^{-1}}^T \Theta^T  
\end{align}
or
\begin{align}
    \hat{W}(t+1) = \hat{W}(t) + \gamma((y - \hat{y})^T C {A^{-1}}^T \Theta^T ) - \eta \left\| e_y\right\| \hat{W} )  
\end{align}

The weight error $\tilde{W} = W - \hat{W}$ and state variable error $\tilde{x} = x-\hat{x}$
\\
Using the system (2) and observer (5), the estimation error dynamics is rewritten as
\\
\begin{align}
     \tilde{\dot{x}} = (PA - N + NQC - LC)x_t + N \tilde{x} + (PB - M)u_t \\- PB f(x,t)_i - G \hat{f}(x,t)_i 
\end{align}
   
\\
\begin{align}
     \tilde{\dot{x}} = (PA -KC)\tilde{x} - PB( f(x,t)_i-\hat{f}(x,t)_i )
\end{align}


\section{Distributed system}

\assumption{Each car $i$ has perfect knowledge only of its own dynamics, i.e. $\tau_i$ is known . The relation $\tau_j = \tau_i + d_{\tau_i} $ will be used . Here $d_{\tau}$ is unknown, but the upper and lower bounds are known. Because when a car connects to his neighborhood, some basic information like which car is share, so $\tau_j$ can be defined base on his type of car } 

So we can write down the the virtual global system that represents all the k-neighbors of $i^th$ . 

\begin{figure}[htp]
    \centering
    \includegraphics[scale=0.45]{img/Spar_Global_new.png}
    \caption{Schematic diagram of the estimation method.}
    \label{fig:information_boarcard}
\end{figure}

\begin{equation}\label{CACC_dis_neibor}
        \mathbf{\hat{X}}(t+1)=\mathbf{A} \mathbf{X} + \mathbf{B}({U+\mathcal{F} + \mathcal{D}_{uk}}) 
\end{equation}

where 
\begin{equation*}
\mathbf{A} = I_N \otimes A_i, \quad \mathbf{B} = I_N \otimes B_i,
\end{equation*}




So the 

\begin{itemize}
\item $\mathbf{X} = [x_{i-k},..., x_{i} ,..., x_{i+k}]^T$ 
\item $U = [u_i , ... u_i ,.. u_i ]^T$ 

 \item $\mathcal{F} = [f_{i-k},..., f_{i} ,..., f_{i+k} ]$ 

 \item $\mathcal{D}_{uk}$ is the uncertain in the controller input. 
 \end{itemize}



So the Global System's state can be ordered by position and have lead, follower, last . 

\remark{It is difficult to get, or send the control input because it will be private. So we will consider that all the controller in vehicle $k^th$ that is the neighbor of $i^th$ vehicle will be  $u_k = u_i + d_{u_k}$ , so the controller in the virtual global system  $U = [u_{i} , ... u_i ,.. u_{i} ]^T$ }

\assumption{In the virtual global system don't fault the system , only fault in the mesurement caused by the uncertain of the communications channels could be considered}

k-neighbor of $i^th$ will send the estimated value $\hat{x}_k $ to the $i^th$ vehicle and , also send about the realtive position from lidar . The measurement of the virtual global system can be :
\begin{equation*}\label{CACC}
      \mathbf{Y}_i = \mathbf{C}_i X    
\end{equation*}  

Where $\mathbf{Y}_i = [\hat{x}_{i-k} ,..., \hat{x}_{i},..., \hat{x}_{i+k} ]^T$  
$e_s$ is the cyber-attack  

\begin{equation}
\mathbf{C}_i=\left[\begin{array}{ccc}
 & I_N \otimes C_i & \\
\mathbf{1}_{\left|\mathcal{N}_i\right| \times 1} & \mathbf{-1}_{\left|\mathcal{N}_i\right| \times 1} & \mathbf{0}]
\end{array}\right]
\end{equation}

where $\mathbf{1}_{\left|\mathcal{N}_i\right|-1 }$
is diagonal vector is 1 , being the $\left|\mathcal{N}_i\right|-1$ dimensional ligne vector 

For exemple 
\begin{equation}
    \mathbf{Y}_2 = \begin{bmatrix}
         C & 0 & 0    \\
         0 & C & 0   \\
         0 & 0 & C   \\
         \mathbf{1} & -\mathbf{1} & 0    \\
         0 & \mathbf{1} & -\mathbf{1}      
     \end{bmatrix}
     \begin{bmatrix}
     x_1 \\
    x_2 \\
    x_3 \\
    \end{bmatrix}
\end{equation}

$\mathbf{1} = [1,0,0]$ and $\mathbf{-1} = [-1,0,0]$   

\section{Observer for virtual global matrix}
We can rewrite the sum of nonlinear and uncertain in the controller to : 
    $\mathcal{G} = \mathcal{F} + \mathcal{D}_{uk}$ 

So in the normal condition each vehicle will moving with the same velocity , and maintain L distance desire . So the dynamic of each vehicle will have a small error . 

So we can use Proportial Intergral Observer for this : 
\begin{equation}\label{obser_global}
    \left\{ \begin{array}{l}
        \mathbf{\hat{X}}(t+1)=\mathbf{A} \mathbf{\hat{X}} + \mathbf{B} U + H\mathcal{\hat{G}} + \mathbf{L}(\mathbf{Y} - \mathbf{C}\mathbf{X} )
        \\ 
         \mathcal{\hat{G}}(t+1) =\mathcal{\hat{G}} +  K(\mathbf{Y} - \mathbf{C}\mathbf{X})
     \end{array}\right.
\end{equation}
 Then, the estimation errors are defined as
 $e_{\hat{x}} (k) = x(k) - \hat{x}_i (k), \quad e_G (k) = \mathcal{G}(k) - \hat{\mathcal{G}} (k) $
 
 Therefore, the dynamics of the estimation errors can be : 
\begin{equation}
    e_{\hat{x}_i} (k + 1) = (\mathbf{A} -\mathbf{LC}) e_{\hat{x}} (k) + He_G (k)
\end{equation}
